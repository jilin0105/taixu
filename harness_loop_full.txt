   455	        dangling.forEach { call ->
   456	            val replayable = !interrupted &&
   457	                ToolReplayPolicy.forTool(call.tool, call.rawToolName) == top.wkbin.taixu.harness.operation.ReplayPolicy.SAFE
   458	            val result = if (replayable) {
   459	                logAgentEvent(sessId, "ToolReplay", "重放中断的只读工具 ${call.rawToolName ?: call.tool.name.lowercase()}")
   460	                try {
   461	                    toolExecutor.execute(call, sessId, workspace)
   462	                } catch (cancellation: CancellationException) {
   463	                    throw cancellation
   464	                } catch (throwable: Throwable) {
   465	                    ToolResult(
   466	                        id = newId(),
   467	                        createdAt = now(),
   468	                        toolCallId = call.id,
   469	                        success = false,
   470	                        output = "重放中断的只读工具失败：${friendly(throwable)}",
   471	                    )
   472	                }
   473	            } else {
   474	                val note = if (interrupted) {
   475	                    "用户停止了本次执行，工具被中断。"
   476	                } else {
   477	                    "工具执行期间应用进程中断。该工具不可自动重放，未再次执行；如仍需要请重新发起。"
   478	                }
   479	                ToolResult(
   480	                    id = newId(),
   481	                    createdAt = now(),
   482	                    toolCallId = call.id,
   483	                    success = false,
   484	                    output = note,
   485	                )
   486	            }
   487	            append(sessId, result)
   488	        }
   489	    }

   491	    fun cancel(targetSessionId: String? = null) {
   492	        val sessId = targetSessionId?.ifBlank { null } ?: _currentSessionId.value
   493	        _sessionPendingMessages[sessId]?.value = emptyList()
   494	        loopScope.launch {
   495	            PromptQueue.entries.forEach { promptQueueManager.clear(sessId, it) }
   496	            refreshPendingProjection(sessId)
   497	        }
   498	        setStatus(sessId, "正在停止…")
   499	        // Do NOT remove the job from the map here: cancellation is asynchronous, and
   500	        // removing by key would let a dying job's finally later delete a *new* job.
   501	        // finishRun removes only its own job via sessionJobs.remove(sessId, selfJob).
   502	        sessionJobs[sessId]?.cancel()
   503	        setSessionState(sessId, SessionRunState.IDLE)
   504	        if (sessId == _currentSessionId.value) {
   505	            _pendingMessages.value = emptyList()
   506	            _status.value = "正在停止…"
   507	            _running.value = false
   508	        }
   509	    }

   540	    private suspend fun finishRun(sessId: String, job: Job) {
   541	        // Only remove ourselves; never clobber a newer job that started after cancel().
   542	        sessionJobs.remove(sessId, job)
   543	        _sessionThinkingLives[sessId]?.value = false
   544	        val waitingApproval = _sessionRunStates.value[sessId] == SessionRunState.WAITING_APPROVAL
   545	        if (!waitingApproval) setStatus(sessId, null)
   546	        if (sessId == _currentSessionId.value) {
   547	            _running.value = false
   548	            if (!waitingApproval) _status.value = null
   549	            _thinkingLive.value = false
   550	        }
   551	        if (waitingApproval) return
   552	        if (tombstonedSessions.contains(sessId)) return
   553	        val (queueItemId, next) = promptQueueManager.first(sessId, PromptQueue.NEXT_RUN) ?: return
   554	        val userMessage = UserMessage(newId(), now(), next.text, next.imageUrls)
   555	        val operationId = operationCoordinator.acceptQueuedRun(sessId, queueItemId, userMessage)
   556	        publishPersistedMessage(sessId, userMessage)
   557	        refreshPendingProjection(sessId)
   558	        startSessionRun(sessId) { runLoopInternal(sessId, now(), operationId) }
   559	    }

   566	    /**
   567	     * Atomically check-and-occupy the session slot under a per-session Mutex,
   568	     * then run [block] as the single active run. If the session is busy the
   569	     * optional [enqueueOnBusy] message is queued for ordered execution.
   570	     */
   571	    private fun startSessionRun(
   572	        sessId: String,
   573	        enqueueOnBusy: PendingMessage? = null,
   574	        block: suspend () -> RunResult,
   575	    ) {
   576	        if (tombstonedSessions.contains(sessId)) return
   577	        loopScope.launch {
   578	            val mutex = sessionMutexes.getOrPut(sessId) { Mutex() }
   579	            var queueAfterUnlock: PendingMessage? = null
   580	            mutex.withLock {
   581	                if (tombstonedSessions.contains(sessId)) return@withLock
   582	                if (isSessionBusy(sessId)) {
   583	                    queueAfterUnlock = enqueueOnBusy
   584	                    return@withLock
   585	                }
   586	                val job = launch(start = CoroutineStart.LAZY) {
   587	                    executeSessionRun(sessId, block)
   588	                }
   589	                sessionJobs[sessId] = job
   590	                job.start()
   591	            }
   592	            queueAfterUnlock?.let { enqueuePending(sessId, it) }
   593	        }
   594	    }
   595	
   596	    /**
   597	     * Claim the session slot unconditionally. Used by approval resumption, which
   598	     * already holds an exclusive claim via claimPending() and is the legitimate
   599	     * successor to a WAITING_APPROVAL run (which still reports busy).
   600	     */
   601	    private fun startClaimedSessionRun(sessId: String, block: suspend () -> RunResult) {
   602	        if (tombstonedSessions.contains(sessId)) return
   603	        loopScope.launch {
   604	            val mutex = sessionMutexes.getOrPut(sessId) { Mutex() }
   605	            mutex.withLock {
   606	                if (tombstonedSessions.contains(sessId)) return@withLock
   607	                val job = launch(start = CoroutineStart.LAZY) {
   608	                    executeSessionRun(sessId, block)
   609	                }
   610	                sessionJobs[sessId] = job
   611	                job.start()
   612	            }
   613	        }
   614	    }
   615	
   616	    private suspend fun executeSessionRun(sessId: String, block: suspend () -> RunResult) {
   617	        val selfJob = currentCoroutineContext()[Job]!!
   618	        setSessionState(sessId, SessionRunState.RUNNING)
   619	        setError(sessId, null)
   620	        try {
   621	            when (val result = block()) {
   622	                RunResult.Completed -> {
   623	                    operationCoordinator.finish(sessId, "completed", getOrCreateLiveMessages(sessId).value.lastOrNull()?.id)
   624	                    setSessionState(sessId, SessionRunState.COMPLETED)
   625	                }
   626	                RunResult.WaitingApproval -> setSessionState(sessId, SessionRunState.WAITING_APPROVAL)
   627	                RunResult.Cancelled -> {
   628	                    operationCoordinator.finish(sessId, "aborted")
   629	                    setSessionState(sessId, SessionRunState.IDLE)
   630	                }
   631	                is RunResult.Failed -> {
   632	                    operationCoordinator.finish(sessId, "failed", details = result.message)
   633	                    setError(sessId, result.message)
   634	                    setSessionState(sessId, SessionRunState.FAILED)
   635	                }
   636	            }
   637	        } catch (cancellation: CancellationException) {
   638	            withContext(NonCancellable) {
   639	                repairDanglingToolCalls(sessId, interrupted = true)
   640	                operationCoordinator.finish(sessId, "aborted", details = "cancelled")
   641	            }
   642	            logger.i("Harness loop cancelled for session $sessId")
   643	            setSessionState(sessId, SessionRunState.IDLE)
   644	        } catch (_: ApprovalPauseException) {
   645	            setSessionState(sessId, SessionRunState.WAITING_APPROVAL)
   646	        } catch (throwable: Throwable) {
   647	            logger.e("Harness loop failed for session $sessId", throwable)
   648	            setError(sessId, throwable.message ?: "执行失败")
   649	            runCatching { operationCoordinator.finish(sessId, "failed", details = throwable.message) }
   650	            setSessionState(sessId, SessionRunState.FAILED)
   651	        } finally {
   652	            finishRun(sessId, selfJob)
   653	        }
   654	    }

   656	    private suspend fun runLoop(sessId: String, userText: String, imageUrls: List<String> = emptyList()): RunResult {
   657	        logAgentEvent(sessId, "UserPrompt", userText)
   658	        val userMessage = UserMessage(id = newId(), createdAt = now(), text = userText, imageUrls = imageUrls)
   659	        val operationId = operationCoordinator.acceptRun(sessId, userMessage)
   660	        publishPersistedMessage(sessId, userMessage)
   661	        return runLoopInternal(sessId, startedAt = now(), operationId = operationId)
   662	    }
   663	
   664	    private suspend fun runLoopInternal(sessId: String, startedAt: Long, operationId: String? = null): RunResult {
   665	        val activeOperationId = operationId ?: operationCoordinator.beginRun(sessId)
   666	        val maxRounds = runCatching { settingsDataStore.maxToolRounds.first() }.getOrDefault(MAX_ROUNDS)
   667	        val autoCwd = runCatching { settingsDataStore.autoWorkspaceCwd.first() }.getOrDefault(true)
   668	        val sessionEntity = sessionDao.findById(sessId)
   669	        val sessionWorkspace = sessionEntity?.workspace.orEmpty()
   670	        // 悬空调用修复须带 workspace：SAFE 工具在此重放，读操作需要正确的工作目录
   671	        repairDanglingToolCalls(sessId, interrupted = false, workspace = sessionWorkspace)
   672	
   673	        val maxToolsPerRound = runCatching { settingsDataStore.maxToolsPerRound.first() }.getOrDefault(12)
   674	        val maxConsecutiveFailures = runCatching { settingsDataStore.maxConsecutiveFailures.first() }.getOrDefault(8)
   675	        val retryPolicy = RetryPolicy.NETWORK_DEFAULT
   676	        var consecutiveFailures = 0
   677	
   678	        var round = 0
   679	        while (round < maxRounds) {
   680	            drainSteeringMessages(sessId)
   681	            setStatus(sessId, "思考中")
   682	            val model = try {
   683	                providerClient.resolveConfigured()
   684	            } catch (cancellation: CancellationException) {
   685	                throw cancellation
   686	            } catch (throwable: Throwable) {
   687	                logAgentEvent(sessId, "ModelResolveError", "无法获取模型配置", throwable)
   688	                appendFatal(sessId, "无法获取模型配置：${friendly(throwable)}", now() - startedAt)
   689	                return RunResult.Failed("无法获取模型配置：${friendly(throwable)}")
   690	            }
   691	            logAgentEvent(sessId, "ModelRequest", "Round=$round, Model=${model.name}, Provider=${model.provider}")
   692	            val msgs = getOrCreateLiveMessages(sessId).value
   693	            val latestUserMessage = msgs.filterIsInstance<UserMessage>().lastOrNull()
   694	            val latestUserText = latestUserMessage?.text.orEmpty()
   695	            val mentionedNames = extractMentionedNames(latestUserText)
   696	            val effectiveModel = if (mentionedNames.isNotEmpty()) {
   697	                val matchedTools = model.dynamicMcpTools.filter { tool ->
   698	                    val sName = tool.serverName.lowercase()
   699	                    val sId = tool.serverId.lowercase()
   700	                    val tName = tool.name.lowercase()
   701	                    sName in mentionedNames || sId in mentionedNames || tName in mentionedNames
   702	                }
   703	                if (matchedTools.isNotEmpty()) model.copy(dynamicMcpTools = matchedTools) else model
   704	            } else {
   705	                model
   706	            }
   707	            appendCapabilityEvents(sessId, latestUserMessage?.id.orEmpty(), mentionedNames, effectiveModel)
   708	            val assistantId = newId()
   709	            val assistantAt = now()
   710	            val streamText = StringBuilder()
   711	            val streamReasoning = StringBuilder()
   712	            var streamed: ChatResult? = null
   713	            var netRetry = 0
   714	            var lastReasoningFlushTime = 0L
   715	            var lastTextFlushTime = 0L
   716	            while (streamed == null) {
   717	                try {
   718	                    operationCoordinator.providerIntent(
   719	                        operationId = activeOperationId,
   720	                        effectId = assistantId,
   721	                        round = round,
   722	                        attempt = netRetry + 1,
   723	                        maxAttempts = retryPolicy.maxAttempts,
   724	                    )
   725	                    streamed = providerClient.chatStream(
   726	                        effectiveModel,
   727	                        apiMessages(sessId, effectiveModel),
   728	                        onReasoning = { chunk ->
   729	                            streamReasoning.append(chunk)
   730	                            sessionThinkingModes[sessId] = true
   731	                            getOrCreateThinkingLiveFlow(sessId).value = true
   732	                            if (sessId == _currentSessionId.value) {
   733	                                _thinkingLive.value = true
   734	                            }
   735	                            val now = System.currentTimeMillis()
   736	                            if (now - lastReasoningFlushTime >= STREAM_FLUSH_INTERVAL_MS) {
   737	                                lastReasoningFlushTime = now
   738	                                streamAssistantReasoning(sessId, assistantId, assistantAt, streamReasoning.toString())
   739	                            }
   740	                        },
   741	                    ) { chunk ->
   742	                        setStatus(sessId, "回复中")
   743	                        streamText.append(chunk)
   744	                        val now = System.currentTimeMillis()
   745	                        if (now - lastTextFlushTime >= STREAM_FLUSH_INTERVAL_MS) {
   746	                            lastTextFlushTime = now
   747	                            streamAssistant(sessId, assistantId, assistantAt, streamText.toString())
   748	                        }
   749	                    }
   750	                    // 流式传输完毕，无条件刷新一次完整内容
   751	                    if (streamReasoning.isNotEmpty()) {
   752	                        streamAssistantReasoning(sessId, assistantId, assistantAt, streamReasoning.toString())
   753	                    }
   754	                    if (streamText.isNotEmpty()) {
   755	                        streamAssistant(sessId, assistantId, assistantAt, streamText.toString())
   756	                    }
   757	                } catch (cancellation: CancellationException) {
   758	                    logAgentEvent(sessId, "Cancelled", "用户主动取消执行")
   759	                    throw cancellation
   760	                } catch (rateLimit: LlmRateLimitException) {
   761	                    currentCoroutineContext().ensureActive()
   762	                    if (rateLimit.quotaExhausted) {
   763	                        getOrCreateThinkingLiveFlow(sessId).value = false
   764	                        if (sessId == _currentSessionId.value) _thinkingLive.value = false
   765	                        logAgentEvent(sessId, "QuotaExhausted", rateLimit.message.orEmpty(), rateLimit)
   766	                        appendFatal(
   767	                            sessId,
   768	                            "模型服务商额度已耗尽，无法继续执行。请充值、切换可用模型或更新 API Key。\n\n${rateLimit.message}",
   769	                            now() - startedAt,
   770	                        )
   771	                        return RunResult.Failed("模型服务商额度已耗尽，无法继续执行。")
   772	                    }
   773	                    netRetry++
   774	                    if (netRetry > retryPolicy.maxRetries) throw rateLimit
   775	                    getOrCreateThinkingLiveFlow(sessId).value = false
   776	                    if (sessId == _currentSessionId.value) _thinkingLive.value = false
   777	                    val waitSeconds = rateLimit.retryAfterSeconds ?: (netRetry * RETRY_BACKOFF_SEC).coerceAtMost(60L)
   778	                    setStatus(sessId, "请求受限，${waitSeconds} 秒后自动重试（$netRetry/$MAX_STREAM_RETRIES）")
   779	                    logAgentEvent(sessId, "RateLimitRetry", "限流退避 ${waitSeconds}s，重试 $netRetry/$MAX_STREAM_RETRIES", rateLimit)
   780	                    streamText.clear()
   781	                    streamReasoning.clear()
   782	                    streamAssistant(sessId, assistantId, assistantAt, "")
   783	                    for (remaining in waitSeconds downTo 1L) {
   784	                        currentCoroutineContext().ensureActive()
   785	                        setStatus(sessId, "请求受限，${remaining} 秒后自动重试（$netRetry/$MAX_STREAM_RETRIES）")
   786	                        delay(1000L)
   787	                    }
   788	                } catch (io: IOException) {
   789	                    currentCoroutineContext().ensureActive()
   790	                    netRetry++
   791	                    logAgentEvent(sessId, "NetworkRetry", "网络中断重试 $netRetry/$MAX_STREAM_RETRIES: ${io.message}", io)
   792	                    if (netRetry > retryPolicy.maxRetries) throw io
   793	                    getOrCreateThinkingLiveFlow(sessId).value = false
   794	                    if (sessId == _currentSessionId.value) {
   795	                        _thinkingLive.value = false
   796	                    }
   797	                    setStatus(sessId, "网络中断，重试中（$netRetry/$MAX_STREAM_RETRIES）")
   798	                    streamText.clear()
   799	                    streamReasoning.clear()
   800	                    streamAssistant(sessId, assistantId, assistantAt, "")
   801	                    delay(retryPolicy.delayForRetry(netRetry))
   802	                } catch (throwable: Throwable) {
   803	                    getOrCreateThinkingLiveFlow(sessId).value = false
   804	                    if (sessId == _currentSessionId.value) {
   805	                        _thinkingLive.value = false
   806	                    }
   807	                    logAgentEvent(sessId, "ModelError", "LLM 调用失败: ${throwable.message}", throwable)
   808	                    if (streamText.isNotEmpty()) {
   809	                        persistAssistant(
   810	                            sessId,
   811	                            assistantId,
   812	                            assistantAt,
   813	                            streamText.toString(),
   814	                            streamReasoning.toString().ifBlank { null },
   815	                            totalMs = now() - startedAt,
   816	                            operationId = activeOperationId,
   817	                            round = round,
   818	                        )
   819	                    } else {
   820	                        appendFatal(sessId, "执行遇到问题，已中断：${friendly(throwable)}", now() - startedAt)
   821	                    }
   822	                    return RunResult.Failed(friendly(throwable))
   823	                }
   824	            }
   825	            val result = streamed
   826	            val jsonMode = model.toolCallMode == ToolCallMode.JSON_TEXT
   827	            val rawText = streamText.toString()
   828	            // JSON 文本模式：从回复文本中解析 [[tool_call]]{...}[[/tool_call]] 标记，
   829	            // 展示给用户与持久化时剥离标记（标记仅作为模型↔引擎的调用协议）
   830	            val jsonCalls = if (jsonMode) extractJsonToolCalls(rawText) else emptyList()
   831	            val displayText = if (jsonMode) stripToolCallMarkers(rawText) else rawText
   832	            logAgentEvent(
   833	                sessId,
   834	                "ModelResponse",
   835	                "TextLength=${rawText.length}, ReasoningLength=${result.reasoningContent?.length ?: 0}, ToolCallsCount=${result.toolCalls.size}, JsonTextCalls=${jsonCalls.size}",
   836	            )
   837	            if (displayText.isNotEmpty()) {
   838	                persistAssistant(
   839	                    sessId,
   840	                    assistantId,
   841	                    assistantAt,
   842	                    displayText,
   843	                    result.reasoningContent,
   844	                    totalMs = if (result.toolCalls.isEmpty() && jsonCalls.isEmpty()) now() - startedAt else null,
   845	                    operationId = activeOperationId,
   846	                    round = round,
   847	                    usage = result.usage,
   848	                    model = effectiveModel,
   849	                )
   850	            } else {
   851	                val usageEntity = result.usage.takeIf { it.hasData }?.let {
   852	                    operationCoordinator.usageEntity(
   853	                        sessionId = sessId,
   854	                        operationId = activeOperationId,
   855	                        entryId = null,
   856	                        provider = effectiveModel.provider,
   857	                        modelId = effectiveModel.model,
   858	                        usage = it,
   859	                    )
   860	                }
   861	                operationCoordinator.providerSettled(activeOperationId, null, usage = usageEntity, round = round)
   862	            }
   863	            getOrCreateThinkingLiveFlow(sessId).value = false
   864	            if (sessId == _currentSessionId.value) {
   865	                _thinkingLive.value = false
   866	            }
   867	            val allCalls = result.toolCalls + jsonCalls
   868	            if (allCalls.isEmpty()) {
   869	                val followUps = promptQueueManager.consume(sessId, PromptQueue.FOLLOW_UP)
   870	                followUps.forEach { publishPersistedMessage(sessId, it) }
   871	                if (followUps.isEmpty()) return RunResult.Completed
   872	                round++
   873	                continue
   874	            }
   875	            // 单轮工具数上限：超出部分回填空结果并提示模型，避免一次性爆发失控。
   876	            val effectiveCalls = if (allCalls.size > maxToolsPerRound) {
   877	                val dropped = allCalls.size - maxToolsPerRound
   878	                allCalls.drop(maxToolsPerRound).forEach { spec ->
   879	                    append(
   880	                        sessId,
   881	                        ToolCall(
   882	                            id = spec.id,
   883	                            createdAt = now(),
   884	                            tool = HarnessApiMapper.toolByName(spec.name),
   885	                            args = buildJsonObject {},
   886	                            reasoning = result.reasoningContent,
   887	                            rawToolName = spec.name.trim(),
   888	                        ),
   889	                    )
   890	                    append(
   891	                        sessId,
   892	                        ToolResult(
   893	                            id = newId(),
   894	                            createdAt = now(),
   895	                            toolCallId = spec.id,
   896	                            success = false,
   897	                            output = "本回合工具调用数量（${allCalls.size}）超过单轮上限（$maxToolsPerRound），已跳过本次多余的 $dropped 个调用。" +
   898	                                "请拆分任务、分步调用工具，避免一次性发起过多工具请求。",
   899	                        ),
   900	                    )
   901	                }
   902	                allCalls.take(maxToolsPerRound)
   903	            } else {
   904	                allCalls
   905	            }
   906	            var roundHadSuccess = false
   907	            effectiveCalls.forEach { spec ->
   908	                val parsedArgs = try {
   909	                    json.parseToJsonElement(spec.argumentsJson) as? JsonObject
   910	                        ?: throw IllegalArgumentException("参数不是 JSON 对象")
   911	                } catch (parseError: Throwable) {
   912	                    append(
   913	                        sessId,
   914	                        ToolCall(
   915	                            id = spec.id,
   916	                            createdAt = now(),
   917	                            tool = HarnessApiMapper.toolByName(spec.name),
   918	                            args = buildJsonObject {},
   919	                            reasoning = result.reasoningContent,
   920	                        ),
   921	                    )
   922	                    append(
   923	                        sessId,
   924	                        ToolResult(
   925	                            id = newId(),
   926	                            createdAt = now(),
   927	                            toolCallId = spec.id,

  1419	    private suspend fun apiMessages(sessId: String, model: ModelConfig): List<ApiMessage> {
  1420	        val compactionEnabled = runCatching { settingsDataStore.contextCompactionEnabled.first() }.getOrDefault(true)
  1421	        val budgetTokens = model.contextTokens
  1422	            ?: runCatching { settingsDataStore.contextBudgetTokens.first() }.getOrDefault(128_000)
  1423	        val sessionEntity = sessionDao.findById(sessId)
  1424	        val sessionWorkspace = sessionEntity?.workspace.orEmpty()
  1425	        val thinkingMode = sessionThinkingModes[sessId] ?: false
  1426	        val toolCallMode = if (model.pureChatMode) ToolCallMode.DISABLED else model.toolCallMode
  1427	
  1428	        var compactedContext = compactionManager.project(sessId)
  1429	        var msgs = compactedContext.messages
  1430	        val latestUserText = msgs.filterIsInstance<UserMessage>().lastOrNull()?.text.orEmpty()
  1431	        val mentionedNames = extractMentionedNames(latestUserText)
  1432	
  1433	        val systemPrompt = if (!model.pureChatMode) {
  1434	            buildSystemPrompt(sessionWorkspace, toolCallMode, mentionedNames, sessId)
  1435	        } else {
  1436	            ""
  1437	        }
  1438	        return buildList {
  1439	            if (systemPrompt.isNotEmpty()) {
  1440	                add(ApiMessage(role = "system", content = systemPrompt))
  1441	            }
  1442	            val answeredIds = msgs.filterIsInstance<ToolResult>().mapTo(mutableSetOf()) { it.toolCallId }
  1443	            val toolCallDetails = msgs.filterIsInstance<ToolCall>().associate {
  1444	                it.id to ((it.rawToolName ?: HarnessApiMapper.apiName(it.tool)) to it.args)
  1445	            }
  1446	
  1447	            // 预算驱动的滑动窗口：从最近一轮往回累加 token，超出预算则更早的历史进入压缩态。
  1448	            // 是否裁剪原文只由真实 token 预算决定，不再按用户轮次阈值强制折叠。
  1449	            val computedKeepFromIndex = if (compactionEnabled) {
  1450	                ContextWindowPolicy.computeKeepFromIndex(
  1451	                    msgs,
  1452	                    budgetTokens,
  1453	                    ContextWindowPolicy.estimateTokens(systemPrompt),
  1454	                )
  1455	            } else {
  1456	                0
  1457	            }
  1458	            if (computedKeepFromIndex > 0) {
  1459	                compactedContext = compactionManager.compact(sessId, compactedContext, computedKeepFromIndex)
  1460	                msgs = compactedContext.messages
  1461	            }
  1462	            val shouldCompact = !compactedContext.summary.isNullOrBlank()
  1463	            val recentTurnCutoffIndex = 0
  1464	
  1465	            if (shouldCompact) {
  1466	                add(
  1467	                    ApiMessage(
  1468	                        role = "system",
  1469	                        content = compactedContext.summary,
  1470	                    ),
  1471	                )
  1472	            }
  1473	
  1474	            // JSON 文本模式：工具调用以文本表达，tool 消息需转成 user 文本（API 不认识 tool 角色）
  1475	            val toolNames = toolCallDetails.mapValuesTo(mutableMapOf()) { it.value.first }
  1476	
  1477	            var i = 0
  1478	            fun apiToolCall(tc: ToolCall) = ApiToolCall(
  1479	                id = tc.id,
  1480	                function = ApiFunctionCall(name = HarnessApiMapper.apiName(tc.tool), arguments = tc.args.toString()),
  1481	            )
  1482	            val isCollapsed = { index: Int -> shouldCompact && index < recentTurnCutoffIndex }
  1483	            while (i < msgs.size) {
  1484	                val message = msgs[i]
  1485	                if (isCollapsed(i)) {
  1486	                    i++
  1487	                    continue
  1488	                }
  1489	                if (message is CapabilityEvent) {
  1490	                    i++
  1491	                    continue
  1492	                }
  1493	                if (toolCallMode == ToolCallMode.JSON_TEXT) {
  1494	                    when (message) {
  1495	                        is ToolCall -> {
  1496	                            toolNames[message.id] = message.rawToolName ?: HarnessApiMapper.apiName(message.tool)
  1497	                            i++
  1498	                        }
  1499	                        is ToolResult -> {
  1500	                            val name = toolNames[message.toolCallId] ?: "工具"
  1501	                            val status = if (message.success) "成功" else "失败"
  1502	                            val args = toolCallDetails[message.toolCallId]?.second
  1503	                            val content = if (isCollapsed(i) && message.output.length > ContextWindowPolicy.compactThresholdFor(name)) {
  1504	                                ContextWindowPolicy.compactToolOutput(name, args, message.output, message.success)
  1505	                            } else {
  1506	                                "【工具 $name 执行结果·$status】\n${message.output}"
  1507	                            }
  1508	                            add(ApiMessage(role = "user", content = content))
  1509	                            i++
  1510	                        }
  1511	                        else -> {
  1512	                            val mapped = when {
  1513	                                isCollapsed(i) && message is AssistantText && message.text.length > 120 ->
  1514	                                    ApiMessage(
  1515	                                        role = "assistant",
  1516	                                        content = ContextWindowPolicy.foldMessageText("助手", message.text),
  1517	                                        reasoning_content = null,
  1518	                                    )
  1519	                                isCollapsed(i) && message is UserMessage && message.text.length > 120 ->
  1520	                                    ApiMessage(
  1521	                                        role = "user",
  1522	                                        content = ContextWindowPolicy.foldMessageText("用户", message.text),
  1523	                                        imageUrls = message.imageUrls,
  1524	                                    )
  1525	                                isCollapsed(i) && message is AssistantText ->
  1526	                                    HarnessApiMapper.toApiMessage(message).copy(reasoning_content = null)
  1527	                                else -> HarnessApiMapper.toApiMessage(message)
  1528	                            }
  1529	                            add(if (message is UserMessage && !model.visionEnabled) mapped.copy(imageUrls = emptyList()) else mapped)
  1530	                            i++
  1531	                        }
  1532	                    }
  1533	                    continue
  1534	                }
  1535	                if (message is AssistantText || message is ToolCall) {
  1536	                    if (message is ToolCall && message.id !in answeredIds) {
  1537	                        i++
  1538	                        continue
  1539	                    }
  1540	                    // 预算折叠态：早期 assistant 文本压缩为一行占位，避免撑爆上下文。
  1541	                    val text = if (isCollapsed(i) && message is AssistantText && message.text.length > 120) {
  1542	                        ContextWindowPolicy.foldMessageText("助手", message.text)
  1543	                    } else {
  1544	                        (message as? AssistantText)?.text
  1545	                    }
  1546	                    // collapsed 历史不携带 reasoning：它是执行过程数据，不应成为长期上下文负担。
  1547	                    val reasoning = if (isCollapsed(i)) null else when (message) {
  1548	                        is AssistantText -> message.reasoning
  1549	                        is ToolCall -> message.reasoning
  1550	                    }
  1551	                    val toolCalls = mutableListOf<ApiToolCall>()
  1552	                    if (message is ToolCall) toolCalls.add(apiToolCall(message))
  1553	                    var j = i + 1
  1554	                    while (j < msgs.size && msgs[j] is ToolCall) {
  1555	                        val tc = msgs[j] as ToolCall
  1556	                        if (tc.id in answeredIds) toolCalls.add(apiToolCall(tc))
  1557	                        j++
  1558	                    }
  1559	                    add(
  1560	                        ApiMessage(
  1561	                            role = "assistant",
  1562	                            content = text,
  1563	                            reasoning_content = when {
  1564	                                isCollapsed(i) -> null
  1565	                                reasoning != null -> reasoning
  1566	                                thinkingMode -> ""
  1567	                                else -> null
  1568	                            },
  1569	                            tool_calls = toolCalls.takeIf { it.isNotEmpty() },
  1570	                        ),
  1571	                    )
  1572	                    i = j
  1573	                } else if (message is ToolResult) {
  1574	                    val detail = toolCallDetails[message.toolCallId]
  1575	                    val content = if (isCollapsed(i) && message.output.length > ContextWindowPolicy.compactThresholdFor(detail?.first)) {
  1576	                        ContextWindowPolicy.compactToolOutput(detail?.first, detail?.second, message.output, message.success)
  1577	                    } else {
  1578	                        message.output
  1579	                    }
  1580	                    add(
  1581	                        ApiMessage(
  1582	                            role = "tool",
  1583	                            content = content,
  1584	                            tool_call_id = message.toolCallId,
  1585	                        ),
  1586	                    )
  1587	                    i++
  1588	                } else {
  1589	                    // 用户消息在早期历史中同样折叠，仅保留极简占位。
  1590	                    val folded = if (isCollapsed(i) && message is UserMessage && message.text.length > 120) {
  1591	                        ContextWindowPolicy.foldMessageText("用户", message.text)
  1592	                    } else {
  1593	                        null
  1594	                    }
  1595	                    if (folded != null) {
  1596	                        add(ApiMessage(role = "user", content = folded))

  1598	                        val mapped = HarnessApiMapper.toApiMessage(message)
  1599	                        add(if (message is UserMessage && !model.visionEnabled) mapped.copy(imageUrls = emptyList()) else mapped)
  1600	                    }
  1601	                    i++
  1602	                }
  1603	            }
  1604	        }

  1714	    /** Approve or reject a frozen tool call, then resume the same Agent session. */
  1715	    fun resolveApproval(requestId: String, approved: Boolean) {
  1716	        loopScope.launch {
  1717	            val request = approvalRepository.find(requestId) ?: return@launch
  1718	            val sessId = request.sessionId
  1719	            if (request.status != top.wkbin.taixu.core.database.AgentApprovalRequestEntity.STATUS_PENDING) return@launch
  1720	            // The original loop may still be unwinding after it persisted the request.
  1721	            // Wait for it before claiming the session slot.
  1722	            sessionJobs[sessId]?.takeIf { it.isActive }?.join()
  1723	
  1724	            // —— 审批有效性校验：过期 / 参数摘要 / 工作区 / operation 归属 ——
  1725	            // 防止“用户批准的是旧参数、旧环境下的请求，实际执行的却是别的东西”。
  1726	            val invalidation = approvalInvalidation(request)
  1727	            val claimedStatus = when {
  1728	                invalidation != null && request.expiresAt <= now() ->
  1729	                    top.wkbin.taixu.core.database.AgentApprovalRequestEntity.STATUS_EXPIRED
  1730	                invalidation != null ->
  1731	                    top.wkbin.taixu.core.database.AgentApprovalRequestEntity.STATUS_FAILED
  1732	                approved ->
  1733	                    top.wkbin.taixu.core.database.AgentApprovalRequestEntity.STATUS_APPROVED
  1734	                else ->
  1735	                    top.wkbin.taixu.core.database.AgentApprovalRequestEntity.STATUS_REJECTED
  1736	            }
  1737	            if (!approvalRepository.claimPending(request.id, claimedStatus)) return@launch
  1738	
  1739	            // Approval resumption is the legitimate successor to a WAITING_APPROVAL run;
  1740	            // claim the slot unconditionally (that state still reports busy to senders).
  1741	            startClaimedSessionRun(sessId) {
  1742	                var approvalResultPersisted = false
  1743	                try {
  1744	                    val result = if (invalidation != null) {
  1745	                        ToolResult(
  1746	                            id = newId(),
  1747	                            createdAt = now(),
  1748	                            toolCallId = request.toolCallId,
  1749	                            success = false,
  1750	                            output = "$invalidation。该工具调用未执行；如仍需要，请重新发起。",
  1751	                        )
  1752	                    } else if (approved) {
  1753	                        val args = json.parseToJsonElement(request.argumentsJson) as? JsonObject
  1754	                            ?: error("审批参数不是 JSON 对象")
  1755	                        val tool = HarnessApiMapper.toolByName(request.toolName)
  1756	                        toolExecutor.execute(
  1757	                            ToolCall(request.toolCallId, request.createdAt, tool, args, rawToolName = request.toolName),
  1758	                            sessId,
  1759	                            request.workspace,
  1760	                            bypassApproval = true,
  1761	                        )
  1762	                    } else {
  1763	                        ToolResult(
  1764	                            id = newId(),
  1765	                            createdAt = now(),
  1766	                            toolCallId = request.toolCallId,
  1767	                            success = false,
  1768	                            output = "用户拒绝了该工具操作。请尊重用户决定，并选择不需要该权限的替代方案。",
  1769	                        )
  1770	                    }
  1771	                    val activeOperation = operationCoordinator.active(sessId)
  1772	                    if (activeOperation != null) {
  1773	                        operationCoordinator.toolSettled(activeOperation.id, result, round = 0)
  1774	                        publishPersistedMessage(sessId, result)
  1775	                    } else {
  1776	                        append(sessId, result)
  1777	                    }
  1778	                    if (invalidation == null) {
  1779	                        approvalRepository.mark(
  1780	                            request.id,
  1781	                            if (approved && result.success) top.wkbin.taixu.core.database.AgentApprovalRequestEntity.STATUS_EXECUTED
  1782	                            else if (approved) top.wkbin.taixu.core.database.AgentApprovalRequestEntity.STATUS_FAILED
  1783	                            else top.wkbin.taixu.core.database.AgentApprovalRequestEntity.STATUS_REJECTED,
  1784	                        )
  1785	                    }
  1786	                    approvalResultPersisted = true
  1787	                    runLoopInternal(sessId, startedAt = now())
  1788	                } catch (cancellation: CancellationException) {
  1789	                    if (!approvalResultPersisted) {
  1790	                        withContext(NonCancellable) {
  1791	                            approvalRepository.mark(
  1792	                                request.id,
  1793	                                if (approved) top.wkbin.taixu.core.database.AgentApprovalRequestEntity.STATUS_FAILED
  1794	                                else top.wkbin.taixu.core.database.AgentApprovalRequestEntity.STATUS_REJECTED,
  1795	                            )
  1796	                            repairDanglingToolCalls(sessId, interrupted = true)
  1797	                        }
  1798	                    }
  1799	                    throw cancellation
  1800	                } catch (throwable: Throwable) {
  1801	                    if (!approvalResultPersisted) {
  1802	                        approvalRepository.mark(request.id, top.wkbin.taixu.core.database.AgentApprovalRequestEntity.STATUS_FAILED)
  1803	                        append(
  1804	                            sessId,
  1805	                            ToolResult(
  1806	                                id = newId(),
  1807	                                createdAt = now(),
  1808	                                toolCallId = request.toolCallId,
  1809	                                success = false,
  1810	                                output = "批准操作执行失败：${friendly(throwable)}",
  1811	                            ),
  1812	                        )
  1813	                    }
  1814	                    RunResult.Failed(throwable.message ?: "审批操作执行失败")
  1815	                }
  1816	            }
  1817	        }
  1818	    }

  1820	    /**
  1821	     * 审批恢复执行前的四重校验；返回 null 表示有效，非 null 为拒绝原因
  1822	     * （会作为 ToolResult 写回，让模型知晓未执行的理由并重新发起）。
  1823	     */
  1824	    private suspend fun approvalInvalidation(
  1825	        request: top.wkbin.taixu.core.database.AgentApprovalRequestEntity,
  1826	    ): String? {
  1827	        if (request.expiresAt <= now()) {
  1828	            val ttlMinutes = ApprovalPolicyEngine.APPROVAL_TTL_MS / 60_000L
  1829	            return "该审批已过期（等待超过 $ttlMinutes 分钟）"
  1830	        }
  1831	        if (request.argsHash.isNotBlank() && request.argsHash != ApprovalPolicyEngine.argsHash(request.argumentsJson)) {
  1832	            return "审批记录的参数摘要校验不一致，审批可能已损坏"
  1833	        }
  1834	        val currentWorkspace = sessionDao.findById(request.sessionId)?.workspace.orEmpty()
  1835	        if (currentWorkspace != request.workspace) {
  1836	            return "会话工作区已变更（审批时：${request.workspace.ifBlank { "无" }}，当前：${currentWorkspace.ifBlank { "无" }}）"
  1837	        }
  1838	        val boundOperationId = request.operationId
  1839	        if (boundOperationId != null && !operationCoordinator.operationExists(boundOperationId)) {
  1840	            return "该审批所属的运行已结束或被新运行接管"
  1841	        }
  1842	        return null
  1843	    }

  1845	    private suspend fun appendCapabilityEvents(
  1846	        sessId: String,
  1847	        userMessageId: String,
  1848	        mentionedNames: Set<String>,
  1849	        model: ModelConfig,
  1850	    ) {
  1851	        if (mentionedNames.isEmpty()) return
  1852	        val messages = getOrCreateLiveMessages(sessId).value
  1853	        if (userMessageId.isBlank()) return
  1854	        val skills = runCatching { skillRepository.allSkills.first() }.getOrDefault(emptyList())
  1855	            .filter { skill ->
  1856	                val names = setOf(skill.name.lowercase(), skill.id.lowercase(), skill.triggerCommand?.removePrefix("/")?.lowercase().orEmpty())
  1857	                names.any { it.isNotBlank() && it in mentionedNames }
  1858	            }
  1859	        skills.forEach { skill ->
  1860	            val key = "skill:$userMessageId:${skill.id}"
  1861	            if (messages.filterIsInstance<CapabilityEvent>().none { it.id == key }) {
  1862	                append(
  1863	                    sessId,
  1864	                    CapabilityEvent(key, now(), CapabilityEvent.Kind.SKILL, skill.name, skill.description),
  1865	                )
  1866	            }
  1867	        }
  1868	        val configuredMcp = runCatching { mcpServerRepository.servers.first() }
  1869	            .getOrDefault(emptyList())
  1870	            .map { it.id to it.name }
  1871	        val discoveredMcp = model.dynamicMcpTools.map { it.serverId to it.serverName }
  1872	        (configuredMcp + discoveredMcp)
  1873	            .distinctBy { it.first }
  1874	            .filter { (serverId, serverName) -> serverId.lowercase() in mentionedNames || serverName.lowercase() in mentionedNames }
  1875	            .forEach { (serverId, serverName) ->
  1876	                val key = "mcp:$userMessageId:$serverId"
  1877	                if (messages.filterIsInstance<CapabilityEvent>().none { it.id == key }) {
  1878	                    append(
  1879	                        sessId,
  1880	                        CapabilityEvent(
  1881	                            key,
  1882	                            now(),
  1883	                            CapabilityEvent.Kind.MCP,
  1884	                            serverName,
  1885	                            if (model.dynamicMcpTools.any { it.serverId == serverId }) {
  1886	                                "MCP 工具已挂载，模型可按需调用"
  1887	                            } else {
  1888	                                "已选择 MCP 服务，正在尝试发现工具；若服务离线，后续会显示连接错误"
  1889	                            },
  1890	                        ),
  1891	                    )
  1892	                }
  1893	            }
  1894	    }
